window.googletag = window.googletag || { cmd: [] };

googletag.cmd.push(function () {

  // Header: desktop grande, tablet leaderboard, mobile banner
  var mapHeader = googletag.sizeMapping()
    .addSize([970, 0], [[970, 250], [970, 90], [728, 90]])
    .addSize([728, 0], [[728, 90]])
    .addSize([0,   0], [[320, 100], [300, 100], [320, 50], [300, 50]])
    .build();

  // Leaderboard: desktop 728/970, mobile banner
  var mapLeaderboard = googletag.sizeMapping()
    .addSize([970, 0], [[970, 90], [728, 90]])
    .addSize([728, 0], [[728, 90]])
    .addSize([0,   0], [[320, 100], [300, 100], [320, 50], [300, 50]])
    .build();

  // Sidebar: desktop 300x600, mobile 300x250
  var mapSidebar = googletag.sizeMapping()
    .addSize([768, 0], [[300, 600], [300, 250]])
    .addSize([0,   0], [[300, 250]])
    .build();

  // Retângulo: 300x250 e 336x280 em qualquer tela
  var mapRetangulo = googletag.sizeMapping()
    .addSize([0, 0], [[300, 250], [336, 280]])
    .build();

  // Mobile only: aparece só em telas menores que 768px
  var mapMobile = googletag.sizeMapping()
    .addSize([768, 0], [])
    .addSize([0,   0], [[320, 100], [300, 100], [320, 50], [300, 50]])
    .build();

  // Header — topo da página
  googletag.defineSlot('/7542/parceiros/{{CODE_NAME}}',
    [[970, 250], [970, 90], [728, 90], [320, 100], [300, 100], [320, 50], [300, 50]],
    'ad-header')
    .defineSizeMapping(mapHeader)
    .addService(googletag.pubads());

  // Conteúdo — meio do texto
  googletag.defineSlot('/7542/parceiros/{{CODE_NAME}}',
    [[300, 250], [336, 280]],
    'ad-conteudo')
    .defineSizeMapping(mapRetangulo)
    .addService(googletag.pubads());

  // Sidebar — barra lateral
  googletag.defineSlot('/7542/parceiros/{{CODE_NAME}}',
    [[300, 600], [300, 250]],
    'ad-sidebar')
    .defineSizeMapping(mapSidebar)
    .addService(googletag.pubads());

  // Rodapé
  googletag.defineSlot('/7542/parceiros/{{CODE_NAME}}',
    [[728, 90], [970, 90], [320, 100], [300, 100], [320, 50], [300, 50]],
    'ad-rodape')
    .defineSizeMapping(mapLeaderboard)
    .addService(googletag.pubads());

  // Mobile only — banner exclusivo mobile
  googletag.defineSlot('/7542/parceiros/{{CODE_NAME}}',
    [[320, 100], [300, 100], [320, 50], [300, 50]],
    'ad-mobile')
    .defineSizeMapping(mapMobile)
    .addService(googletag.pubads());

  googletag.setConfig({
    singleRequest: true,
    collapseEmptyDivs: true,
    centering: true
  });
  googletag.enableServices();
});
</script>
